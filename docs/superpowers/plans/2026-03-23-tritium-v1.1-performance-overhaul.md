### Task 2: Fix broken AVX2 batch_dot accumulator in simd.h

**Files:**

- [ ] **Step 1: Rewrite the AVX2 path in batch_dot**

In `engine/include/tritium/simd.h`, replace the `batch_dot` function (lines 50-135) with:
```cpp
inline int batch_dot(const TritWord64* a, const TritWord64* b, size_t n_words) {
    int result = 0;

#if defined(__AVX2__)
    size_t w = 0;
    const __m256i even_mask = _mm256_set1_epi64x(
        static_cast<long long>(TritWord64::EVEN_BITS));

    for (; w + 4 <= n_words; w += 4) {
        __m256i va = _mm256_loadu_si256(reinterpret_cast<const __m256i*>(&a[w]));
        __m256i vb = _mm256_loadu_si256(reinterpret_cast<const __m256i*>(&b[w]));

        __m256i a_lo = _mm256_and_si256(va, even_mask);
        __m256i a_hi = _mm256_and_si256(_mm256_srli_epi64(va, 1), even_mask);
        __m256i b_lo = _mm256_and_si256(vb, even_mask);
        __m256i b_hi = _mm256_and_si256(_mm256_srli_epi64(vb, 1), even_mask);

        __m256i a_nz = _mm256_or_si256(a_lo, a_hi);
        __m256i b_nz = _mm256_or_si256(b_lo, b_hi);
        __m256i both_nz = _mm256_and_si256(a_nz, b_nz);

        __m256i a_neg = _mm256_and_si256(a_hi, a_lo);
        __m256i b_neg = _mm256_and_si256(b_hi, b_lo);

        __m256i sign_diff = _mm256_and_si256(_mm256_xor_si256(a_neg, b_neg), both_nz);
        __m256i sign_agr = _mm256_andnot_si256(sign_diff, both_nz);

        // Extract and popcount (AVX2 lacks native popcount)
        uint64_t agr_vals[4], dif_vals[4];
        _mm256_storeu_si256(reinterpret_cast<__m256i*>(agr_vals), sign_agr);
        _mm256_storeu_si256(reinterpret_cast<__m256i*>(dif_vals), sign_diff);
        for (int i = 0; i < 4; ++i)
            result += popcount64(agr_vals[i]) - popcount64(dif_vals[i]);
    }

    // Scalar tail
    for (; w < n_words; ++w) {
        uint64_t va = a[w].bits, vb = b[w].bits;
        uint64_t a_lo = va & TritWord64::EVEN_BITS;
        uint64_t a_hi = (va >> 1) & TritWord64::EVEN_BITS;
        uint64_t b_lo = vb & TritWord64::EVEN_BITS;
        uint64_t b_hi = (vb >> 1) & TritWord64::EVEN_BITS;
        uint64_t a_nz = a_lo | a_hi, b_nz = b_lo | b_hi;
        uint64_t both_nz = a_nz & b_nz;
        uint64_t a_neg = a_hi & a_lo, b_neg = b_hi & b_lo;
        uint64_t sd = (a_neg ^ b_neg) & both_nz;
        uint64_t sa = both_nz & ~sd;
        result += popcount64(sa) - popcount64(sd);
    }
#else
    for (size_t w = 0; w < n_words; ++w) {
        uint64_t va = a[w].bits, vb = b[w].bits;
        uint64_t a_lo = va & TritWord64::EVEN_BITS;
        uint64_t a_hi = (va >> 1) & TritWord64::EVEN_BITS;
        uint64_t b_lo = vb & TritWord64::EVEN_BITS;
        uint64_t b_hi = (vb >> 1) & TritWord64::EVEN_BITS;
        uint64_t a_nz = a_lo | a_hi, b_nz = b_lo | b_hi;
        uint64_t both_nz = a_nz & b_nz;
        uint64_t a_neg = a_hi & a_lo, b_neg = b_hi & b_lo;
        uint64_t sd = (a_neg ^ b_neg) & both_nz;
        uint64_t sa = both_nz & ~sd;
        result += popcount64(sa) - popcount64(sd);
    }
#endif

    return result;
}
```

The key fix: removed the broken `acc_agree`/`acc_differ` accumulators and the pointless `_mm256_or_si256` + `_mm256_slli_epi64(..., 0)` lines. Now directly stores and popcounts per iteration.

```cpp
#pragma once
#include "trit.h"
#include "simd.h"
#include <cstddef>
#include <cassert>

namespace tritium {

// Non-owning view into a span of packed TritWord64s.
// Lightweight (pointer + 2 size_t). Same pattern as std::span.
// WARNING: Must not outlive the TritVector/TritMatrix it references.
class TritVectorView {
    TritWord64* data_;
    size_t size_;
    size_t num_words_;

public:
    TritVectorView() : data_(nullptr), size_(0), num_words_(0) {}
    TritVectorView(TritWord64* data, size_t size, size_t num_words)
        : data_(data), size_(size), num_words_(num_words) {}

    size_t size() const { return size_; }
    size_t num_words() const { return num_words_; }
    const TritWord64* data() const { return data_; }
    TritWord64* data() { return data_; }

    Trit operator[](size_t i) const {
        assert(i < size_);
        return data_[i / TritWord64::CAPACITY].get(
            static_cast<int>(i % TritWord64::CAPACITY));
    }

    void set(size_t i, Trit t) {
        assert(i < size_);
        data_[i / TritWord64::CAPACITY].set(
            static_cast<int>(i % TritWord64::CAPACITY), t);
    }

    int dot(const TritVectorView& other) const {
        assert(size_ == other.size_);
        return simd::batch_dot(data_, other.data_, num_words_);
    }

    int hamming(const TritVectorView& other) const {
        assert(size_ == other.size_);
        int count = 0;
        for (size_t w = 0; w < num_words_; ++w) {
            uint64_t diff = data_[w].bits ^ other.data_[w].bits;
            uint64_t diff_lo = diff & TritWord64::EVEN_BITS;
            uint64_t diff_hi = (diff >> 1) & TritWord64::EVEN_BITS;
            count += popcount64(diff_lo | diff_hi);
        }
        return count;
    }

    int l0_norm() const {
        int count = 0;
        for (size_t w = 0; w < num_words_; ++w)
            count += data_[w].popcount_nonzero();
        return count;
    }
};

} // namespace tritium
```

```cpp
#pragma once
#include "trit_view.h"
#include "trit_vector.h"
#include "packing.h"
#include "simd.h"
#include <vector>
#include <cstddef>
#include <cassert>

namespace tritium {

class TritMatrix {
    std::vector<TritWord64> data_;
    size_t rows_count_;
    size_t cols_count_;
    size_t words_per_row_;

public:
    TritMatrix() : rows_count_(0), cols_count_(0), words_per_row_(0) {}

    TritMatrix(size_t rows, size_t cols)
        : rows_count_(rows), cols_count_(cols),
          words_per_row_(word_count(cols)),
          data_(rows * word_count(cols)) {}

    TritMatrix(const Trit* data, size_t rows, size_t cols)
        : TritMatrix(rows, cols) {
        for (size_t r = 0; r < rows; ++r)
            for (size_t c = 0; c < cols; ++c)
                set(r, c, data[r * cols + c]);
    }

    // Move semantics
    TritMatrix(TritMatrix&&) = default;
    TritMatrix& operator=(TritMatrix&&) = default;
    TritMatrix(const TritMatrix&) = default;
    TritMatrix& operator=(const TritMatrix&) = default;

    size_t rows() const { return rows_count_; }
    size_t cols() const { return cols_count_; }

    Trit get(size_t r, size_t c) const {
        return data_[r * words_per_row_ + c / TritWord64::CAPACITY].get(
            static_cast<int>(c % TritWord64::CAPACITY));
    }

    void set(size_t r, size_t c, Trit t) {
        data_[r * words_per_row_ + c / TritWord64::CAPACITY].set(
            static_cast<int>(c % TritWord64::CAPACITY), t);
    }

    TritVectorView row(size_t r) {
        return TritVectorView(&data_[r * words_per_row_], cols_count_, words_per_row_);
    }

    TritVectorView row(size_t r) const {
        return TritVectorView(
            const_cast<TritWord64*>(&data_[r * words_per_row_]),
            cols_count_, words_per_row_);
    }

    std::vector<int> matvec(const TritVector& v) const {
        assert(v.size() == cols_count_);
        std::vector<int> result(rows_count_);
        for (size_t r = 0; r < rows_count_; ++r)
            result[r] = simd::batch_dot(
                &data_[r * words_per_row_], v.data(), words_per_row_);
        return result;
    }

    std::vector<std::vector<int>> matmul(const TritMatrix& B) const {
        assert(cols_count_ == B.rows_count_);
        TritMatrix BT = B.transpose();
        std::vector<std::vector<int>> result(rows_count_,
            std::vector<int>(B.cols_count_, 0));
        for (size_t i = 0; i < rows_count_; ++i)
            for (size_t j = 0; j < B.cols_count_; ++j)
                result[i][j] = simd::batch_dot(
                    &data_[i * words_per_row_],
                    &BT.data_[j * BT.words_per_row_],
                    words_per_row_);
        return result;
    }

    TritMatrix transpose() const {
        TritMatrix result(cols_count_, rows_count_);
        for (size_t r = 0; r < rows_count_; ++r)
            for (size_t c = 0; c < cols_count_; ++c)
                result.set(c, r, get(r, c));
        return result;
    }
};

} // namespace tritium
```

```cpp
inline std::vector<int> conv1d(const TritVector& input, const TritVector& kernel) {
    assert(input.size() >= kernel.size());
    size_t out_len = input.size() - kernel.size() + 1;
    std::vector<int> output(out_len, 0);
    size_t ksize = kernel.size();

    // Pre-extract kernel values once
    std::vector<int> k_vals(ksize);
    for (size_t k = 0; k < ksize; ++k)
        k_vals[k] = to_int(kernel[k]);

    // Main loop with direct word access
    const TritWord64* in_words = input.data();
    for (size_t i = 0; i < out_len; ++i) {
        int sum = 0;
        for (size_t k = 0; k < ksize; ++k) {
            size_t idx = i + k;
            int a = to_int(in_words[idx / TritWord64::CAPACITY].get(
                static_cast<int>(idx % TritWord64::CAPACITY)));
            sum += a * k_vals[k];
        }
        output[i] = sum;
```

```cpp
#pragma once
#include <cstddef>
#include <cstdlib>
#include <new>

namespace tritium {

template<typename T, size_t Alignment = 64>
struct AlignedAllocator {
    using value_type = T;

    AlignedAllocator() = default;
    template<typename U> AlignedAllocator(const AlignedAllocator<U, Alignment>&) {}

    T* allocate(size_t n) {
        size_t bytes = n * sizeof(T);
#if defined(_MSC_VER)
        void* p = _aligned_malloc(bytes, Alignment);
#else
        void* p = std::aligned_alloc(Alignment, (bytes + Alignment - 1) & ~(Alignment - 1));
#endif
        if (!p) throw std::bad_alloc();
        return static_cast<T*>(p);
    }

    void deallocate(T* p, size_t) {
#if defined(_MSC_VER)
        _aligned_free(p);
#else
        std::free(p);
#endif
    }

    template<typename U> bool operator==(const AlignedAllocator<U, Alignment>&) const { return true; }
    template<typename U> bool operator!=(const AlignedAllocator<U, Alignment>&) const { return false; }
};

} // namespace tritium
```
