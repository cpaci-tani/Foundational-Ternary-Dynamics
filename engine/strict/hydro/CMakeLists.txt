cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(ftd_hydro_candidate LANGUAGES CXX)
  if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
  endif()
endif()
add_library(ftd_hydro_candidate STATIC hydro_runtime.cpp)
target_compile_features(ftd_hydro_candidate PUBLIC cxx_std_17)
target_include_directories(ftd_hydro_candidate PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(ftd_hydro_candidate PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(ftd_hydro_candidate PRIVATE /W4 /permissive-)
else()
  target_compile_options(ftd_hydro_candidate PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(ftd_hydro_cli main.cpp)
target_link_libraries(ftd_hydro_cli PRIVATE ftd_hydro_candidate)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  enable_testing()
endif()
add_test(NAME hydro_candidate_native_selftest COMMAND ftd_hydro_cli --selftest)
option(FTD_HYDRO_CUDA "Build optional hydro successor CUDA kernels" OFF)
option(FTD_HYDRO_WASM "Build optional hydro successor WASM bindings" OFF)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/cuda.cmake")
  include("${CMAKE_CURRENT_SOURCE_DIR}/cuda.cmake")
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/wasm.cmake")
  include("${CMAKE_CURRENT_SOURCE_DIR}/wasm.cmake")
endif()
