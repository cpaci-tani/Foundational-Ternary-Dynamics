cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(ftd_strict_candidate LANGUAGES CXX)
  if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
  endif()
endif()
add_library(ftd_strict_candidate STATIC staged_runtime.cpp)
target_compile_features(ftd_strict_candidate PUBLIC cxx_std_17)
target_include_directories(ftd_strict_candidate PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(ftd_strict_candidate PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(ftd_strict_candidate PRIVATE /W4 /permissive-)
else()
  target_compile_options(ftd_strict_candidate PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(ftd_strict_cli main.cpp)
target_link_libraries(ftd_strict_cli PRIVATE ftd_strict_candidate)
add_executable(ftd_strict_selftest native_selftest.cpp)
target_link_libraries(ftd_strict_selftest PRIVATE ftd_strict_candidate)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  enable_testing()
endif()
add_test(NAME strict_candidate_native_atomicity COMMAND ftd_strict_selftest)
include("${CMAKE_CURRENT_SOURCE_DIR}/cuda.cmake" OPTIONAL)
include("${CMAKE_CURRENT_SOURCE_DIR}/wasm.cmake" OPTIONAL)
