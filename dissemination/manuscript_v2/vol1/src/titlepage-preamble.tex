\newcommand{\ornamentalline}{%

      \draw[goldthread, line width=0.5pt] (0,0) -- (2,0);
      \fill[goldthread] (2.2,0) circle (2pt);
      \draw[goldthread, line width=0.5pt] (2.4,0) -- (4.4,0);

      \draw[goldthread, line width=0.5pt] (4.8,0) -- (6.8,0);

}
